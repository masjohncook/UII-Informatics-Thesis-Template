% Kotak petunjuk berwarna merah seperti penanda pada template Word.
\newcommand{\petunjuk}[1]{%
  \begin{center}
  \setlength{\fboxsep}{8pt}%
  \fcolorbox{instructionborder}{instructionbg}{%
    \parbox{0.92\textwidth}{\color{instructionborder}\singlespacing\fontsize{10}{12}\selectfont\textbf{PETUNJUK TEMPLATE:} #1}%
  }
  \end{center}
}
