     \immediate\write\txtfile{#2 ..... #3}%   
   \else
     \ifx\tmp\levelsec
       \immediate\write\txtfile{\space\space #2 ...... #3}%
     \else
       %%% Code for further toc levels
     \fi
   \fi
  }
\begin{document}
\input{\toc.toc}
